import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid stagger>
  <Card title="Feedback while you write" icon="pencil">
    The Compass app in the Astro dev toolbar reviews the page you are viewing
    and flags drift as soon as you save.
  </Card>
  <Card title="Audits in CI" icon="approve-check">
    Opt-in audits [fail pull requests](/guides/audit-in-ci/) that turn a
    tutorial into a reference page. Deployments never call an API.
  </Card>
  <Card title="Nine built-in rules" icon="seti:folder">
    Classify pages with [Diátaxis](/reference/rules/#diataxis), rate them
    against [quality rubrics](/reference/rules/#quality), find
    [duplicates](/reference/rules/#duplicates) and check the frontmatter of
    plugins like `starlight-tags` and `starlight-blog`.
  </Card>
  <Card title="Provider-agnostic" icon="puzzle">
    Use [TypeSafe's Jev](/reference/providers/#typesafe),
    [OpenAI Decisions](/reference/providers/#openaidecisions) or
    [any OpenAI-compatible model](/reference/providers/#openaicompatible), or write your own
    [provider or rule](/concepts/how-it-works/#providers-and-rules).
  </Card>
  <Card title="Helpful for readers" icon="open-book">
    Show [documentation type badges](/guides/documentation-type-badges/), let
    readers [ask the docs](/guides/ask-the-docs/) from a sparkle button without
    hallucinations, and [triage their feedback](/guides/triage-feedback/).
  </Card>
</CardGrid>
